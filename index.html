💻 Model Tanımı (Özet)

# xPass++ GAT — 3 Katman, 4 Kafa, 128 Gizli Boyut class xPassPlusPlusGAT(torch.nn.Module): def __init__(self, in_channels=16, hidden=128, heads=4): self.gat1 = GATConv(in_channels, hidden, heads=heads, dropout=0.3) self.gat2 = GATConv(hidden*heads, hidden, heads=heads, dropout=0.3) self.gat3 = GATConv(hidden*heads, hidden, heads=1, dropout=0.3) self.dqs_head = Linear(hidden, 1) # → Decision Quality Score def forward(self, x, edge_index): x = F.elu(self.gat1(x, edge_index)) x = F.elu(self.gat2(x, edge_index)) x = F.elu(self.gat3(x, edge_index)) return torch.sigmoid(self.dqs_head(x)) # [0, 1] DQS çıktısı
